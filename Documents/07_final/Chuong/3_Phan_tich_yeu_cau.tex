\section{Thiết kế Lớp (Class Design)}
Phần này mô tả cấu trúc thiết kế hướng đối tượng lớp chi tiết cho hệ thống đi chợ tiện lợi \textbf{NaviMart}. Mô hình thiết kế lớp được xây dựng dựa trên nguyên lý BCE (Boundary - Control - Entity) kết hợp với kiến trúc phân lớp nhằm tối ưu tính độc lập và khả năng bảo trì.

Hệ thống được tổ chức thành 8 mô-đun chính:
\begin{enumerate}
    \item \textbf{Mô-đun Xác thực (Auth Module):} Quản lý tài khoản người dùng, băm mật khẩu, phân quyền qua JWT và OTP qua Email.
    \item \textbf{Mô-đun Nhóm gia đình (Family Module):} Liên kết tài khoản các thành viên trong gia đình để chia sẻ dữ liệu đi chợ và thực phẩm.
    \item \textbf{Mô-đun Lập Danh sách Mua sắm (Shopping Lists Module):} Quản lý quy trình lập danh sách đi chợ và tiến độ mua sắm thực tế.
    \item \textbf{Mô-đun Kho Thực phẩm (Pantry Module):} Quản lý thực phẩm hiện có trong tủ lạnh, vị trí bảo quản và cảnh báo hạn sử dụng.
    \item \textbf{Mô-đun Kế hoạch Ăn uống (Meals Module):} Lên lịch trình ăn uống hàng ngày và theo dõi bữa ăn.
    \item \textbf{Mô-đun Công thức nấu ăn (Recipes Module):} Kho công thức nấu ăn và gợi ý thực đơn thông minh bằng trí tuệ nhân tạo.
    \item \textbf{Mô-đun Thống kê và Báo cáo (Reports Module):} Phân tích chi tiêu, biểu đồ xu hướng tiêu thụ thực phẩm và lượng thực phẩm lãng phí.
    \item \textbf{Mô-đun Quản trị Hệ thống (Admin Module):} Phê duyệt công thức, kiểm duyệt người dùng và quản lý danh mục thực phẩm chuẩn.
\end{enumerate}

\subsection{Cấu trúc Phân lớp BCE}
Mỗi mô-đun chức năng tuân thủ nghiêm ngặt các vai trò phân lớp sau:
\begin{itemize}
    \item \textbf{Lớp Boundary (Views \& API Clients):}
    \begin{itemize}
        \item \textbf{Views (React Components):} Đóng vai trò là thành phần giao diện người dùng hiển thị HTML/CSS, nhận sự kiện tương tác của người dùng.
        \item \textbf{API Clients (Axios Clients):} Chịu trách nhiệm đóng gói dữ liệu và thực hiện các yêu cầu HTTP (GET, POST, PATCH, DELETE) đồng bộ dữ liệu với Backend.
    \end{itemize}
    \item \textbf{Lớp Control (Controllers \& Services):}
    \begin{itemize}
        \item \textbf{Controllers (NestJS Controllers):} Tiếp nhận các yêu cầu HTTP từ Client, kiểm tra tính hợp lệ dữ liệu đầu vào (DTO Validation) và điều hướng xuống tầng nghiệp vụ.
        \item \textbf{Services (NestJS Services):} Nơi tập trung toàn bộ các quy tắc nghiệp vụ (Business Logic), thuật toán phân tích, tính toán, và tương tác với cơ sở dữ liệu.
    \end{itemize}
    \item \textbf{Lớp Entity (Database Entities):}
    \begin{itemize}
        \item \textbf{Entities (TypeORM/Prisma Entities):} Ánh xạ trực tiếp cấu trúc các bảng cơ sở dữ liệu vật lý thành các đối tượng mã nguồn để thực hiện các thao tác CRUD.
    \end{itemize}
\end{itemize}

\section{Đặc tả Giao tiếp Hệ thống (System Interface Design)}
Hệ thống \textbf{NaviMart} áp dụng giao tiếp REST API đồng bộ bằng dữ liệu JSON để đảm bảo sự gắn kết lỏng lẻo và khả năng mở rộng. Tất cả các yêu cầu gửi đến các endpoint bảo mật yêu cầu tiêu đề xác thực:
\begin{verbatim}
Authorization: Bearer <JWT_Token>
\end{verbatim}

\subsection{Các Dịch vụ Giao tiếp Bên ngoài (External Interfaces)}
Hệ thống tích hợp 3 cổng giao tiếp dịch vụ bên thứ ba chính:
\begin{enumerate}
    \item \textbf{Google OAuth 2.0 API:} Hỗ trợ đăng nhập một chạm từ phía Client, kiểm tra và xác thực danh tính người dùng thông qua Google Identity Services.
    \item \textbf{Nodemailer \& SMTP Service (Email API):} Gửi email mã OTP xác thực, lấy lại mật khẩu và cảnh báo định kỳ về thực phẩm sắp hết hạn đến hòm thư cá nhân của người dùng.
    \item \textbf{OpenAI Chat Completion API (GPT-4o):} Kết nối bảo mật bằng API Key từ NestJS Service để truyền nội dung hội thoại, ngữ cảnh kho thực phẩm hiện có để AI sinh câu trả lời gợi ý thực đơn cá nhân hóa thời gian thực.
\end{enumerate}

\subsection{Danh sách các Bảng Dữ liệu Vật lý}
Cơ sở dữ liệu NaviMart bao gồm 10 bảng dữ liệu vật lý chính:
\begin{enumerate}
    \item \textbf{\texttt{users}:} Lưu trữ tài khoản, email, mật khẩu băm, phân quyền và liên kết nhóm.
    \item \textbf{\texttt{family\_groups}:} Quản lý thông tin nhóm gia đình, mã mời và chủ sở hữu.
    \item \textbf{\texttt{categories}:} Phân loại các nhóm thực phẩm (ví dụ: Rau củ, Hải sản).
    \item \textbf{\texttt{food\_catalog}:} Từ điển thực phẩm chuẩn hệ thống để chuẩn hóa dữ liệu đơn vị tính.
    \item \textbf{\texttt{shopping\_lists}:} Lịch sử và kế hoạch các danh sách mua sắm.
    \item \textbf{\texttt{shopping\_list\_items}:} Chi tiết số lượng thực phẩm cần mua và trạng thái đã mua.
    \item \textbf{\texttt{inventory}:} Thực phẩm tồn kho trong tủ lạnh gia đình kèm vị trí bảo quản và hạn sử dụng.
    \item \textbf{\texttt{recipes}:} Công thức chế biến món ăn do người dùng/admin đóng góp.
    \item \textbf{\texttt{recipe\_ingredients}:} Định lượng nguyên liệu yêu cầu cho từng công thức món ăn (bảng trung gian).
    \item \textbf{\texttt{meal\_plans}:} Kế hoạch ăn uống của gia đình theo ngày và theo buổi ăn.
\end{enumerate}

\subsection{Kịch bản khởi tạo Cơ sở dữ liệu vật lý (SQL Schema)}
\begin{lstlisting}[style=mysqlstyle, caption=SQL Schema Definition for NaviMart]
-- Kich ban khoi tao he thong co so du lieu cua NaviMart

CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    email VARCHAR(150) NOT NULL UNIQUE,
    phone VARCHAR(20),
    password_hash VARCHAR(255) NOT NULL,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    dob DATE NOT NULL,
    gender VARCHAR(10) NOT NULL,
    role VARCHAR(20) NOT NULL,
    group_id INT,
    status VARCHAR(20) NOT NULL
);

CREATE TABLE family_groups (
    group_id INT AUTO_INCREMENT PRIMARY KEY,
    group_name VARCHAR(100) NOT NULL,
    owner_id INT NOT NULL,
    invite_code VARCHAR(50) UNIQUE
);

CREATE TABLE categories (
    category_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL
);

CREATE TABLE food_catalog (
    food_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(150) NOT NULL,
    category_id INT NOT NULL,
    unit VARCHAR(20) NOT NULL,
    FOREIGN KEY (category_id) REFERENCES categories(category_id)
);

CREATE TABLE shopping_lists (
    list_id INT AUTO_INCREMENT PRIMARY KEY,
    group_id INT NOT NULL,
    name VARCHAR(100) NOT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    type VARCHAR(20),
    FOREIGN KEY (group_id) REFERENCES family_groups(group_id)
);

CREATE TABLE shopping_list_items (
    item_id INT AUTO_INCREMENT PRIMARY KEY,
    list_id INT NOT NULL,
    food_id INT NOT NULL,
    quantity DECIMAL(10,2) NOT NULL,
    status VARCHAR(20) DEFAULT 'Pending',
    FOREIGN KEY (list_id) REFERENCES shopping_lists(list_id),
    FOREIGN KEY (food_id) REFERENCES food_catalog(food_id)
);

CREATE TABLE inventory (
    inventory_id INT AUTO_INCREMENT PRIMARY KEY,
    group_id INT NOT NULL,
    food_id INT NOT NULL,
    quantity DECIMAL(10,2) NOT NULL,
    expiry_date DATE NOT NULL,
    location VARCHAR(50) NOT NULL,
    FOREIGN KEY (group_id) REFERENCES family_groups(group_id),
    FOREIGN KEY (food_id) REFERENCES food_catalog(food_id)
);

CREATE TABLE recipes (
    recipe_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(200) NOT NULL,
    instructions TEXT NOT NULL,
    image_url VARCHAR(255),
    author_id INT NOT NULL,
    status VARCHAR(20) DEFAULT 'Pending',
    FOREIGN KEY (author_id) REFERENCES users(user_id)
);

CREATE TABLE recipe_ingredients (
    recipe_id INT NOT NULL,
    food_id INT NOT NULL,
    quantity_required DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (recipe_id, food_id),
    FOREIGN KEY (recipe_id) REFERENCES recipes(recipe_id),
    FOREIGN KEY (food_id) REFERENCES food_catalog(food_id)
);

CREATE TABLE meal_plans (
    plan_id INT AUTO_INCREMENT PRIMARY KEY,
    group_id INT NOT NULL,
    date DATE NOT NULL,
    meal_type VARCHAR(20) NOT NULL,
    recipe_id INT NOT NULL,
    is_completed BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (group_id) REFERENCES family_groups(group_id),
    FOREIGN KEY (recipe_id) REFERENCES recipes(recipe_id)
);

-- Thiet lap moi lien ket khoa ngoai cheo nguoi dung - gia dinh
ALTER TABLE users ADD FOREIGN KEY (group_id) REFERENCES family_groups(group_id);
ALTER TABLE family_groups ADD FOREIGN KEY (owner_id) REFERENCES users(user_id);

\subsection{Các Kịch bản Truy vấn Cơ sở dữ liệu Chi tiết}
Dưới đây giải thích cách hoạt động của cơ sở dữ liệu để thực hiện các chức năng chính của hệ thống thông qua các câu lệnh truy vấn SQL:

\subsubsection{Kịch bản 1: Người dùng tham gia vào Nhóm gia đình}
Khi một người dùng nhập mã mời (\texttt{invite\_code}) của một gia đình từ ứng dụng di động:
\begin{enumerate}
    \item Tìm kiếm nhóm gia đình tương ứng với \texttt{invite\_code}.
    \item Cập nhật trường \texttt{group\_id} của người dùng đó (ví dụ \texttt{user\_id = 42}) bằng khóa ngoại tương ứng.
\end{enumerate}

\begin{lstlisting}[style=mysqlstyle, caption=SQL Update User Group on Joining Family]
-- Buoc 1: Kiem tra su ton tai va lay ID nhom gia dinh tu ma moi
SELECT group_id FROM family_groups WHERE invite_code = 'XYZ123';

-- Buoc 2: Cap nhat thong tin nhom cho nguoi dung
UPDATE users 
SET group_id = (SELECT group_id FROM family_groups WHERE invite_code = 'XYZ123')
WHERE user_id = 42;
\end{lstlisting}

\subsubsection{Kịch bản 2: Lọc thực phẩm sắp hết hạn trong tủ lạnh}
Hệ thống chạy dịch vụ quét để cảnh báo cho gia đình (ví dụ nhóm \texttt{group\_id = 5}) về các thực phẩm trong kho chuẩn bị hết hạn (trong vòng 3 ngày tới):

\begin{lstlisting}[style=mysqlstyle, caption=Query Expiring Foods in Pantry]
SELECT 
    i.inventory_id,
    fc.name AS food_name,
    i.quantity,
    fc.unit,
    i.expiry_date,
    i.location,
    DATEDIFF(i.expiry_date, CURDATE()) AS days_left
FROM inventory i
JOIN food_catalog fc ON i.food_id = fc.food_id
WHERE i.group_id = 5 
  AND i.expiry_date BETWEEN CURDATE() AND DATE_ADD(CURDATE(), INTERVAL 3 DAY)
ORDER BY i.expiry_date ASC;
\end{lstlisting}

\subsubsection{Kịch bản 3: Tự động so khớp nguyên liệu còn thiếu để nấu ăn}
Khi người dùng chọn nấu một món ăn theo công thức (\texttt{recipe\_id = 12}), hệ thống đối chiếu định lượng nguyên liệu yêu cầu với kho thực tế (\texttt{inventory} của gia đình \texttt{group\_id = 5}):

\begin{lstlisting}[style=mysqlstyle, caption=SQL Query to Check Missing Ingredients]
SELECT 
    fc.name AS ingredient_name,
    ri.quantity_required AS qty_needed,
    COALESCE(SUM(i.quantity), 0) AS qty_in_pantry,
    fc.unit,
    CASE 
        WHEN COALESCE(SUM(i.quantity), 0) >= ri.quantity_required THEN 'Du nguyen lieu'
        ELSE CONCAT('Thieu ', ri.quantity_required - COALESCE(SUM(i.quantity), 0))
    END AS status
FROM recipe_ingredients ri
JOIN food_catalog fc ON ri.food_id = fc.food_id
LEFT JOIN inventory i ON ri.food_id = i.food_id AND i.group_id = 5
WHERE ri.recipe_id = 12
GROUP BY ri.food_id, fc.name, ri.quantity_required, fc.unit;
\end{lstlisting}

\subsubsection{Kịch bản 4: Tự động lập danh sách đi chợ từ kế hoạch bữa ăn}
Khi gia đình có kế hoạch ăn uống cho ngày cụ thể và muốn hệ thống tự động sinh danh sách đi chợ (\texttt{list\_id = 10}) các món còn thiếu:

\begin{lstlisting}[style=mysqlstyle, caption=Insert Missing Ingredients from Meal Plan to Shopping List]
INSERT INTO shopping_list_items (list_id, food_id, quantity, status)
SELECT 
    10 AS list_id,
    needed.food_id,
    (needed.total_needed - COALESCE(pantry.total_owned, 0)) AS quantity_to_buy,
    'Pending' AS status
FROM (
    -- Lay tong luong nguyen lieu can cho tat ca bua an trong ngay 2026-06-25
    SELECT ri.food_id, SUM(ri.quantity_required) AS total_needed
    FROM meal_plans mp
    JOIN recipe_ingredients ri ON mp.recipe_id = ri.recipe_id
    WHERE mp.group_id = 5 AND mp.date = '2026-06-25'
    GROUP BY ri.food_id
) needed
LEFT JOIN (
    -- Lay tong luong thuc pham dang co san trong tu lanh
    SELECT food_id, SUM(quantity) AS total_owned
    FROM inventory
    WHERE group_id = 5
    GROUP BY food_id
) pantry ON needed.food_id = pantry.food_id
WHERE needed.total_needed > COALESCE(pantry.total_owned, 0);
\end{lstlisting}

\subsubsection{Kịch bản 5: Báo cáo tỷ lệ thực phẩm bị lãng phí}
Hệ thống phân tích báo cáo thống kê cho gia đình (\texttt{group\_id = 5}) về tỷ lệ phần trăm lãng phí thực phẩm (các thực phẩm mua về nhưng hết hạn hoặc hỏng, được đánh dấu là \texttt{Wasted}):

\begin{lstlisting}[style=mysqlstyle, caption=SQL Query for Consumption Waste Report]
SELECT 
    fc.name AS food_name,
    SUM(CASE WHEN sli.status = 'Wasted' THEN sli.quantity ELSE 0 END) AS total_wasted,
    SUM(CASE WHEN sli.status = 'Bought' THEN sli.quantity ELSE 0 END) AS total_bought,
    (SUM(CASE WHEN sli.status = 'Wasted' THEN sli.quantity ELSE 0 END) / 
     NULLIF(SUM(CASE WHEN sli.status = 'Bought' THEN sli.quantity ELSE 0 END), 0)) * 100 AS waste_percentage
FROM shopping_list_items sli
JOIN food_catalog fc ON sli.food_id = fc.food_id
JOIN shopping_lists sl ON sli.list_id = sl.list_id
WHERE sl.group_id = 5
GROUP BY fc.food_id, fc.name
HAVING total_bought > 0
ORDER BY waste_percentage DESC;
\end{lstlisting}
